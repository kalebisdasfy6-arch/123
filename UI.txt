local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/sametexe001/sametlibs/refs/heads/main/Thugsense/Library.lua"))()

-- Inject the purple theme colors BEFORE creating the window
if Library.Theme then
    Library.Theme.Accent = Color3.fromRGB(143, 0, 255)
    Library.Theme.Background = Color3.fromRGB(15, 15, 15)
    Library.Theme.Secondary = Color3.fromRGB(22, 22, 22)
    Library.Theme.Stroke = Color3.fromRGB(30, 30, 30)
    Library.Theme.Text = Color3.fromRGB(240, 240, 240)
end

local Window = Library:Window({
    Name = 'miani football  |  [v2.6]  |  License: <font color="#FFAA00">Lifetime</font>  |  <font color="#55FF00">Premium User</font>',
    Size = UDim2.new(0, 500, 0, 600),
    FadeSpeed = 0.0
})

local Watermark = Library:Watermark("miani football")
local KeybindList = Library:KeybindList()
Watermark:SetVisibility(false)
KeybindList:SetVisibility(false)

-- Navigation tabs configuration
local PlayerTab = Window:Page({Name = "Player", Columns = 2, Subtabs = false})
local MainTab = Window:Page({Name = "Main", Columns = 2, Subtabs = false})
local FarmingTab = Window:Page({Name = "Farming", Columns = 2, Subtabs = false})
local AimingTab = Window:Page({Name = "Aiming", Columns = 2, Subtabs = false})
local VisualsTab = Window:Page({Name = "Visuals", Columns = 2, Subtabs = false})
local SettingsTab = Library:CreateSettingsPage(Window, Watermark, KeybindList)

-- Optimize tab switching lag
for _, page in ipairs(Window.Pages) do
    page.Turn = function(self, Bool)
        if self.Active == Bool then return end
        self.Active = Bool
        
        if Bool then 
            self.Elements["Page"].Instance.Visible = true
            self.Elements["Text"].Instance.TextColor3 = Library.Theme.Accent
            self.Elements["Text"].Instance.TextTransparency = 0
            self.Elements["Hide"].Instance.Visible = true
        else
            self.Elements["Page"].Instance.Visible = false
            self.Elements["Text"].Instance.TextColor3 = Library.Theme.Text
            self.Elements["Text"].Instance.TextTransparency = 0.5
            self.Elements["Hide"].Instance.Visible = false
        end
    end
end

-- ============================================================================
-- MOBILE SUPPORT / CLOSE BUTTON
-- ============================================================================
local CoreGui = game:GetService("CoreGui")
local LocalPlayer = game:GetService("Players").LocalPlayer

local mobileGui = Instance.new("ScreenGui")
mobileGui.Name = "MianiFootballMobileToggleMenu"
mobileGui.ResetOnSpawn = false
mobileGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
mobileGui.DisplayOrder = 99999

pcall(function()
    local success = pcall(function() local _ = CoreGui.Name end)
    if success then
        mobileGui.Parent = CoreGui
    else
        mobileGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
    end
end)

local toggleButton = Instance.new("ImageButton")
toggleButton.Size = UDim2.new(0, 68, 0, 68)
toggleButton.Position = UDim2.new(0, 12, 0.5, -34)
toggleButton.Image = "rbxassetid://109948061273706"
toggleButton.BackgroundColor3 = Color3.fromRGB(18, 18, 22)
toggleButton.BackgroundTransparency = 0.15
toggleButton.ScaleType = Enum.ScaleType.Stretch
toggleButton.Active = true
toggleButton.Parent = mobileGui
local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0, 18); corner.Parent = toggleButton
local stroke = Instance.new("UIStroke"); stroke.Color = Color3.fromRGB(255, 0, 0); stroke.Thickness = 3; stroke.Parent = toggleButton
local textBadge = Instance.new("TextLabel")
textBadge.Size = UDim2.new(1, 20, 0, 20); textBadge.Position = UDim2.new(0, -10, 1, 4)
textBadge.BackgroundColor3 = Color3.fromRGB(18, 18, 22); textBadge.Text = "CLOSE"
textBadge.TextColor3 = Color3.fromRGB(255, 255, 255); textBadge.Font = Enum.Font.GothamBold; textBadge.TextSize = 11
textBadge.BorderSizePixel = 0; textBadge.Parent = toggleButton
Instance.new("UICorner", textBadge).CornerRadius = UDim.new(0, 6)
local badgeStroke = Instance.new("UIStroke", textBadge); badgeStroke.Color = Color3.fromRGB(255, 0, 0); badgeStroke.Thickness = 1.5

local uiVisible = true
Library.MenuKeybind = "None"
local function toggleUI()
    uiVisible = not uiVisible
    pcall(function()
        if Window and Window.Elements and Window.Elements["MainFrame"] then
            Window.Elements["MainFrame"].Instance.Visible = uiVisible
        end
    end)
    textBadge.Text = uiVisible and "CLOSE" or "MENU"
    textBadge.BackgroundColor3 = uiVisible and Color3.fromRGB(18, 18, 22) or Color3.fromRGB(150, 20, 20)
end
toggleButton.MouseButton1Click:Connect(toggleUI)

local dragging, dragStart, startPos
toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = input.Position; startPos = toggleButton.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
game:GetService("UserInputService").InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
game:GetService("UserInputService").InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.RightShift or input.KeyCode == Enum.KeyCode.RightControl or input.KeyCode == Enum.KeyCode.Insert then toggleUI() end
end)


-- ============================================================================
-- FINAL INIT
-- ============================================================================
Library:Init()

task.spawn(function()
    task.wait(0.2)
    pcall(function()
        if Window and Window.Elements and Window.Elements["Title"] then
            Window.Elements["Title"].Instance.RichText = true
        end
    end)
end)
