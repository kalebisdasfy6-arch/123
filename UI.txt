        -- Make SetVisible a no-op so NOTHING can toggle the UI except our button
        pcall(function()
            MainWindow.Cfg.SetVisible = function() end
        end)

        -- Kill only the logo-related connections from Aether's connection list
        -- (Aether stores every :Connect() it makes in Library.Connections)
        pcall(function()
            local logo = MainWindow.Items.Logo and MainWindow.Items.Logo.Instance
            if not logo then return end
            logo.Active = false

            -- Find and disconnect any connection whose signal belongs to the logo
            local toRemove = {}
            for i, connInfo in ipairs(Library.Connections or {}) do
                local ok, isSame = pcall(function()
                    return connInfo.Event and
                           (connInfo.Event == logo.InputBegan or
                            connInfo.Event == logo.MouseButton1Click or
                            connInfo.Event == logo.InputEnded)
                end)
                if ok and isSame then
                    pcall(function() connInfo.Connection:Disconnect() end)
                    table.insert(toRemove, 1, i)
                end
            end
            for _, i in ipairs(toRemove) do
                table.remove(Library.Connections, i)
            end
        end)
    end)
end)

-- ============================================================
--  PATCH: Mobile scroll fix — make left/right columns show scrollbar
--  and increase scroll frame touch priority so dragging scrolls, not clicks
-- ============================================================
task.spawn(function()
    task.wait(0.3)
    pcall(function()
        -- Find all ScrollingFrames that Aether uses for Left/Right columns
        -- and make them properly scroll on mobile
        local function patchScrollFrames(root)
            for _, v in ipairs(root:GetDescendants()) do
                if v:IsA("ScrollingFrame") and v.ScrollBarThickness == 0 then
                    v.ScrollBarThickness = 3
                    v.ScrollBarImageColor3 = Color3.fromRGB(61, 220, 132)
                    v.ScrollBarImageTransparency = 0.5
                    v.ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
                    v.ScrollingEnabled = true
                    -- Active=true means it eats touch input for scrolling
                    -- which stops it passing to children while scrolling
                    v.Active = true
                end
            end
        end
        local menu = MainWindow.Items.Menu.Instance
        patchScrollFrames(menu)
        -- Also patch any that get added later (tab switches)
        menu.DescendantAdded:Connect(function(d)
            if d:IsA("ScrollingFrame") and d.ScrollBarThickness == 0 then
                task.wait(0.05)
                d.ScrollBarThickness = 3
                d.ScrollBarImageColor3 = Color3.fromRGB(61, 220, 132)
                d.ScrollBarImageTransparency = 0.5
                d.ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
                d.ScrollingEnabled = true
                d.Active = true
            end
        end)
    end)
end)

-- ============================================================
--  PATCH: Make ALL Aether dropdowns scrollable (click-safe)
-- ============================================================
do
    local _origAddDropdown = Library.AddDropdown
    Library.AddDropdown = function(self, Data)
        local dd = _origAddDropdown(self, Data)
        task.spawn(function()
            task.wait(0.05)
            pcall(function()
                local holder = dd.Items.DropdownHolder.Instance
                -- Just cap the max height — keep AutomaticSize so Aether
                -- still controls it, but clamp it so it doesn't go forever
                local constraint = Instance.new("UISizeConstraint")
                constraint.MaxSize = Vector2.new(math.huge, 180)
                constraint.Parent = holder
                holder.ClipsDescendants = true
                -- Enable scroll wheel on the holder itself
                -- Aether uses a TextButton so we hook InputChanged on it
                local layout = holder:FindFirstChildOfClass("UIListLayout")
                local scrollOffset = 0
                holder.InputChanged:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseWheel then
                        if not layout then return end
                        -- Get total content height
                        local contentHeight = layout.AbsoluteContentSize.Y
                        local visibleHeight = holder.AbsoluteSize.Y
                        if contentHeight <= visibleHeight then return end
                        -- Scroll by moving all children up/down via padding
                        local pad = holder:FindFirstChildOfClass("UIPadding")
                        if not pad then
                            pad = Instance.new("UIPadding")
                            pad.Parent = holder
                        end
                        local step = 30
                        scrollOffset = math.clamp(
                            scrollOffset - (input.Position.Z * step),
                            0,
                            contentHeight - visibleHeight
                        )
                        pad.PaddingTop = UDim.new(0, -scrollOffset)
                    end
                end)
            end)
        end)
        return dd
    end
end

local function sendNotification(title, text)
    pcall(function()
        Library:Notify({ Title = title .. ": " .. (text or ""), Lifetime = 4 })
    end)
    pcall(function()
        StarterGui:SetCore("SendNotification", { Title = title, Text = text or "", Duration = 4 })
    end)
end

local function notifyToggle(name, enabled)
    sendNotification(name, enabled and "enabled" or "disabled")
end

local function formatMoney(amount)
    local formatted = tostring(math.floor(tonumber(amount) or 0))
    while true do
        local n, c = formatted:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
        formatted = n
        if c == 0 then break end
    end
    return "$" .. formatted
end

local function normalizeDropdownValue(v)
    if type(v) == "table" then
        if v[1] then return v[1] end
        for k, en in pairs(v) do if en then return k end end
        return nil
    end
    return v
end

local DupeToolName = nil

local function duplicateCurrentItem()
    if DupeCooldown then sendNotification("Dupe","Cooldown!"); return end
    DupeCooldown = true
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local Backpack = LocalPlayer:WaitForChild("Backpack")
    local Tool = char:FindFirstChildOfClass("Tool")
    if not Tool then sendNotification("Dupe","Hold a tool first!"); DupeCooldown=false; return end
    Tool.Parent = Backpack; task.wait(0.5)
    local ToolName = Tool.Name; local ToolId
    local delay = 0.25 + ((math.clamp(getPing(),0,300)/300)*0.03)
    local conn
    if ReplicatedStorage:FindFirstChild("MarketItems") then
        conn = ReplicatedStorage.MarketItems.ChildAdded:Connect(function(item)
            if item.Name == ToolName then
                local ow = item:WaitForChild("owner",2)
                if ow and ow.Value == LocalPlayer.Name then ToolId = item:GetAttribute("SpecialId") or item:GetAttribute("ToolId") end
            end
        end)
    end
    pcall(function() ReplicatedStorage.ListWeaponRemote:FireServer(ToolName, 676767) end)
    task.wait(delay)
    pcall(function() ReplicatedStorage.BackpackRemote:InvokeServer("Store", ToolName) end)
    task.wait(3)
    if ToolId then pcall(function() ReplicatedStorage.BuyItemRemote:FireServer(ToolName,"Remove",ToolId) end) end
    pcall(function() ReplicatedStorage.BackpackRemote:InvokeServer("Grab", ToolName) end)
    task.wait(1)
    local duped = Backpack:FindFirstChild(ToolName)
    if duped and LocalPlayer.Character then duped.Parent = LocalPlayer.Character end
    if conn then conn:Disconnect() end
    task.wait(2); DupeCooldown = false
end

task.spawn(function()
    while task.wait(0.1) do
        if getgenv().AutoDupeCurrentItem and DupeToolName and not DupeCooldown then
            duplicateCurrentItem()
        end
    end
end)

local function GetNearestCar(hrp)
    local civs = workspace:FindFirstChild("CivCars"); if not civs then return nil end
    local best, bestD = nil, math.huge
    for _, car in ipairs(civs:GetChildren()) do
        if car:IsA("Model") then
            local s = car:FindFirstChild("DriveSeat") or car:FindFirstChildWhichIsA("VehicleSeat")
            if s and not s.Occupant then
                if not car.PrimaryPart then pcall(function() car.PrimaryPart=car:FindFirstChildWhichIsA("BasePart") end) end
                if car.PrimaryPart then
                    local d=(car.PrimaryPart.Position-hrp.Position).Magnitude

        end
    end

local function getSafeFolder()
    local hds=workspace:FindFirstChild("HomeDoorSS"); local hd=hds and hds:FindFirstChild("HomeDoor")
    return hd and hd:FindFirstChild("Safe")
end
local function GetSafeItems()
    local items={}; local inv=LocalPlayer:FindFirstChild("InvData")
    if inv then for _,c in ipairs(inv:GetChildren()) do table.insert(items,c.Name) end end
    table.sort(items); return items
end
local function GetAllBackpackItems()
    local items={}; local char=LocalPlayer.Character; local bp=LocalPlayer:FindFirstChild("Backpack")
    if char then for _,t in pairs(char:GetChildren()) do if t:IsA("Tool") and not table.find(items,t.Name) then table.insert(items,t.Name) end end end
    if bp then for _,t in pairs(bp:GetChildren()) do if t:IsA("Tool") and not table.find(items,t.Name) then table.insert(items,t.Name) end end end
    table.sort(items); return items
end
local function TakeFromSafe(itemName)
    local sf=getSafeFolder(); if not sf then sendNotification("Safe","Safe not found!"); return false end
    local sp=sf:FindFirstChild("Union") or sf:FindFirstChildWhichIsA("BasePart"); if not sp then return false end
    local char=LocalPlayer.Character; local root=char and char:FindFirstChild("HumanoidRootPart"); if not root then return false end
    local orig=root.CFrame
    pcall(function() Config:Teleport(sp.CFrame*CFrame.new(0,2,0)); task.wait(0.3); ReplicatedStorage:FindFirstChild("Inventory"):FireServer("Change",itemName,"Inv",sf); task.wait(0.5); Config:Teleport(orig); task.wait(0.5) end)
    local got=LocalPlayer.Backpack:FindFirstChild(itemName) or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild(itemName))
    sendNotification("Safe", got and ("Took "..itemName) or ("Failed: "..itemName)); return got~=nil
end
local function StoreHoldingItem()
    local char=LocalPlayer.Character; if not char then return end
    local tool=char:FindFirstChildOfClass("Tool"); if not tool then sendNotification("Safe","Hold an item first!"); return end
    local sf=getSafeFolder(); if not sf then sendNotification("Safe","Safe not found!"); return end
    local sp=sf:FindFirstChild("Union") or sf:FindFirstChildWhichIsA("BasePart"); local root=char:FindFirstChild("HumanoidRootPart"); if not sp or not root then return end
    local orig=root.CFrame; local hum=char:FindFirstChildOfClass("Humanoid"); if hum then hum:UnequipTools() end; task.wait(0.2)
    Config:Teleport(sp.CFrame*CFrame.new(0,2,0)); task.wait(0.3)
    pcall(function() if ReplicatedStorage:FindFirstChild("BackpackRemote") then ReplicatedStorage.BackpackRemote:InvokeServer("Store",tool.Name) end; if ReplicatedStorage:FindFirstChild("Inventory") then ReplicatedStorage.Inventory:FireServer("Change",tool.Name,"Backpack",sf) end end)
    task.wait(0.5); Config:Teleport(orig); sendNotification("Safe","Stored "..tool.Name)
end
local autoDupeActive=false; local dupeSpeed=15; local dupeCounter=0; local selectedDupeItem=nil
local function SafeDupeOnce(itemName, returnCFrame)
    local sf=getSafeFolder(); if not sf then return false end
    local sp=sf:FindFirstChild("Union") or sf:FindFirstChildWhichIsA("BasePart"); if not sp then return false end
    pcall(function()
        Config:Teleport(sp.CFrame*CFrame.new(0,2,0)); task.wait(0.3)
        task.spawn(function() if ReplicatedStorage:FindFirstChild("BackpackRemote") then ReplicatedStorage.BackpackRemote:InvokeServer("Store",itemName) end end)
        task.spawn(function() if ReplicatedStorage:FindFirstChild("Inventory") then ReplicatedStorage.Inventory:FireServer("Change",itemName,"Backpack",sf) end end)
        task.wait(0.6); Config:Teleport(returnCFrame); task.wait(1.2)
        if ReplicatedStorage:FindFirstChild("BackpackRemote") then ReplicatedStorage.BackpackRemote:InvokeServer("Grab",itemName) end; task.wait(0.5)
    end)
    return true
end

    corner.Parent = btn

    badge.BorderSizePixel        = 0
    badge.Text                   = "CLOSE"
    badge.TextColor3             = ACCENT
    badge.Font                   = Enum.Font.GothamBold
    badge.TextSize               = 10
    badge.Parent                 = btn

    Instance.new("UICorner", badge).CornerRadius = UDim.new(0, 4)

    local badgeStroke = Instance.new("UIStroke")
    badgeStroke.Color        = ACCENT
    badgeStroke.Thickness    = 1
    badgeStroke.Transparency = 0.4
    badgeStroke.Parent       = badge

    -- Direct toggle: show/hide the entire ScreenGui that Aether uses
    -- This is 100% reliable regardless of IsMobile, tweens, or Aether internals
    local aetherGui = Library.Items and Library.Items.Instance

    local function setVisible(v)
        uiVisible = v
        if aetherGui then aetherGui.Enabled = v end
        badge.Text       = v and "CLOSE" or "OPEN"
        badge.TextColor3 = v and ACCENT or ACCENT_OFF
        stroke.Color     = v and ACCENT or ACCENT_OFF
    end

    task.spawn(function()

            )

        if input.KeyCode == Enum.KeyCode.RightShift
        or input.KeyCode == Enum.KeyCode.RightControl
        or input.KeyCode == Enum.KeyCode.Insert then
            setVisible(not uiVisible)
        end
    end)
end)

    task.spawn(function()
        local r = 0
        while animStroke.Parent do
            r = (r + 2) % 360
            grad.Rotation = r
            task.wait(0.04)
        end
    end)

    -- ── 3. WATERMARK RAINBOW ────────────────────────────────
    pcall(function()
        if Library.HUD and Library.HUD.Instance then
            for _, desc in ipairs(Library.HUD.Instance:GetDescendants()) do
                if desc:IsA("TextLabel") and desc.Text:find("miami football") then
                    task.spawn(function()
                        local h = 0
                        while desc.Parent do
                            h = (h + 0.012) % 1
                            desc.TextColor3 = Color3.fromHSV(h, 0.7, 1)
                            task.wait(0.05)
                        end
                    end)
                    break
                end
            end
        end
    end)

    -- ── 4. TAB BUTTON HOVER GLOW (no TweenService — just color swap, no lag) ──
    pcall(function()
        for _, tab in ipairs({PlayerTab, MainTab, AutofarmTab, VisualsTab, SafeTab}) do
            local btn = tab.Items and tab.Items.Button and tab.Items.Button.Instance
            if not btn then continue end
            btn.MouseEnter:Connect(function()
                btn.BackgroundTransparency = 0.6
            end)
            btn.MouseLeave:Connect(function()
                btn.BackgroundTransparency = 1
            end)
        end
    end)

    -- ── 6. TOGGLE BUTTON PULSE ANIMATION ────────────────────
    pcall(function()
        local toggleGui = game:GetService("CoreGui"):FindFirstChild("MiamiToggleButton")
        if not toggleGui then return end
        local toggleBtn = toggleGui:FindFirstChild("MiamiToggleBtn")
        if not toggleBtn then return end
        local strokeObj = toggleBtn:FindFirstChildOfClass("UIStroke")
        if not strokeObj then return end

        task.spawn(function()
            while toggleBtn.Parent do
                TS:Create(strokeObj, TweenInfo.new(0.9, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Thickness = 3.5}):Play()
                task.wait(0.9)
                TS:Create(strokeObj, TweenInfo.new(0.9, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {Thickness = 1.5}):Play()
                task.wait(0.9)
            end
        end)

        -- Rainbow color cycle on toggle button stroke
        task.spawn(function()
            local h = 0.38  -- start green-ish
            while strokeObj.Parent do
                h = (h + 0.004) % 1
                strokeObj.Color = Color3.fromHSV(h, 1, 1)
                task.wait(0.05)
            end
        end)
    end)

    -- ── 7. SMOOTH TAB SWITCH HIGHLIGHT ──────────────────────
    pcall(function()
        local sideBar = MainWindow.Items.SideBar and MainWindow.Items.SideBar.Instance
        if not sideBar then return end
        -- add a subtle bottom border line in green
        local line = Instance.new("Frame")
        line.Size = UDim2.new(1, 0, 0, 1)
        line.Position = UDim2.new(0, 0, 1, -1)
        line.BackgroundColor3 = GREEN
        line.BackgroundTransparency = 0.6
        line.BorderSizePixel = 0
        line.Parent = sideBar
    end)
end)
