local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0, 18); corner.Parent = toggleButton
local stroke = Instance.new("UIStroke"); stroke.Color = Color3.fromRGB(255, 0, 0); stroke.Thickness = 3; stroke.Parent = toggleButton
local textBadge = Instance.new("TextLabel")
textBadge.Size = UDim2.new(1, 20, 0, 20); textBadge.Position = UDim2.new(0, -10, 1, 4)
textBadge.BackgroundColor3 = Color3.fromRGB(18, 18, 22); textBadge.Text = "CLOSE"
textBadge.TextColor3 = Color3.fromRGB(255, 255, 255); textBadge.Font = Enum.Font.GothamBold; textBadge.TextSize = 11
textBadge.BorderSizePixel = 0; textBadge.Parent = toggleButton
Instance.new("UICorner", textBadge).CornerRadius = UDim.new(0, 6)
local badgeStroke = Instance.new("UIStroke", textBadge); badgeStroke.Color = Color3.fromRGB(255, 0, 0); badgeStroke.Thickness = 1.5

local uiVisible = true
Library.MenuKeybind = "None"
local function toggleUI()
    uiVisible = not uiVisible
    pcall(function()
        if Window and Window.Elements and Window.Elements["MainFrame"] then
            Window.Elements["MainFrame"].Instance.Visible = uiVisible
        end
    end)
    textBadge.Text = uiVisible and "CLOSE" or "MENU"
    textBadge.BackgroundColor3 = uiVisible and Color3.fromRGB(18, 18, 22) or Color3.fromRGB(150, 20, 20)
end
toggleButton.MouseButton1Click:Connect(toggleUI)

local dragging, dragStart, startPos
toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = input.Position; startPos = toggleButton.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
game:GetService("UserInputService").InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
game:GetService("UserInputService").InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.RightShift or input.KeyCode == Enum.KeyCode.RightControl or input.KeyCode == Enum.KeyCode.Insert then toggleUI() end
end)

Library:Init()

task.spawn(function()
    for i = 1, 15 do
        task.wait(0.5)
        pcall(function()
            if Window and Window.Elements and Window.Elements["Title"] then
                Window.Elements["Title"].Instance.RichText = true
            end
            
            local uiScreen = nil
            local success = pcall(function() local _ = CoreGui.Name end)
            if success then
                uiScreen = CoreGui:FindFirstChild("Thugsense")
            else
                uiScreen = LocalPlayer:FindFirstChild("PlayerGui"):FindFirstChild("Thugsense")
            end
            
            if uiScreen then
                local uiScale = uiScreen:FindFirstChildWhichIsA("UIScale", true)
                if uiScale and uiScale.Scale < 1 then uiScale.Scale = 1 end
                
                for _, v in pairs(uiScreen:GetDescendants()) do
                    if v:IsA("TextLabel") and string.find(v.Text, "miani football") then
                        v.RichText = true
                    end
                end
            end
        end)
    end
end)
