local function normalizeDropdownValue(value)
    if type(value) == "table" then
        if value[1] then
            return value[1]
        end
        for key, enabled in pairs(value) do
            if enabled then
                return key
            end
        end
        return nil
    end
    return value
end

local function formatMoney(amount)
    local formatted = tostring(math.floor(tonumber(amount) or 0))
    while true do
        local newFormatted, count = formatted:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
        formatted = newFormatted
        if count == 0 then break end
    end
    return "$" .. formatted
end

local function duplicateCurrentItem()
    if DupeCooldown then
        warn("Cooldown! Please wait.")
        return
    end
    DupeCooldown = true

    local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local Backpack = LocalPlayer:WaitForChild('Backpack')
    local Tool = Character:FindFirstChildOfClass('Tool')
    if not Tool then
        warn("Couldn't find your tool! Please make sure you're holding one.")
        DupeCooldown = false
        return
    end

    Tool.Parent = Backpack
    task.wait(0.5)

    local ToolName = Tool.Name
    local ToolId = nil

    local ping = getPing()
    local delay = 0.25 + ((math.clamp(ping, 0, 300) / 300) * 0.03)

    local Connection = ReplicatedStorage.MarketItems.ChildAdded:Connect(function(item)
        if item.Name == ToolName then
            local owner = item:WaitForChild('owner', 2)
            if owner and owner.Value == LocalPlayer.Name then
                ToolId = item:GetAttribute('SpecialId')
            end
        end
    end)

    task.spawn(function()
        ReplicatedStorage.ListWeaponRemote:FireServer(ToolName, 676767)
    end)

    task.wait(delay)

    task.spawn(function()
        ReplicatedStorage.BackpackRemote:InvokeServer('Store', ToolName)
    end)

    task.wait(3)

    if ToolId then
        task.spawn(function()
            ReplicatedStorage.BuyItemRemote:FireServer(ToolName, 'Remove', ToolId)
        end)
    end

    task.spawn(function()
        ReplicatedStorage.BackpackRemote:InvokeServer('Grab', ToolName)
    end)

    task.wait(1)

    local dupedTool = Backpack:FindFirstChild(ToolName)
    if dupedTool then
        dupedTool.Parent = Character
    end

    Connection:Disconnect()
    task.wait(2)
    DupeCooldown = false
end

local function DropMoneyAmount(Amount)
    if not ReplicatedStorage then return false, "ReplicatedStorage missing" end

    local BankRemote = ReplicatedStorage:FindFirstChild('BankProcessRemote', true)
    if not BankRemote then
        return false, "BankProcessRemote not found"
    end

    local function send(...)
        if BankRemote.InvokeServer then
            return BankRemote:InvokeServer(...)
        elseif BankRemote.FireServer then
            return BankRemote:FireServer(...)
        end
    end

    local stringAmount = tostring(Amount)
    local numberAmount = tonumber(Amount) or Amount

    local attempts = {
        {"Drop", stringAmount},
        {"Drop", numberAmount},
        {{"Drop", stringAmount}},
        {{"Drop", numberAmount}},
        {"Drop", stringAmount, false},
        {"Drop", numberAmount, false},
        {{"Drop", stringAmount, false}},
        {{"Drop", numberAmount, false}},
    }

    for _, args in ipairs(attempts) do
        local ok, result = pcall(send, table.unpack(args))
        if ok and result ~= false then
            return true, result
        end
    end

    return false, "Failed to drop money"
end

-- ProximityPrompt Utilities
local promptConnections = {}
local promptAddedConnection

local function setupPrompt(prompt, enableInstant)
    if not prompt or not prompt:IsA("ProximityPrompt") then return end
    if promptConnections[prompt] then promptConnections[prompt] = nil end
    if enableInstant then
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
    else
        prompt.HoldDuration = 1
        prompt.RequiresLineOfSight = true
    end
    promptConnections[prompt] = prompt
end

local function refreshPrompts(enableInstant)
    for _, prompt in ipairs(workspace:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") then setupPrompt(prompt, enableInstant) end
    end
    if promptAddedConnection then
        promptAddedConnection:Disconnect()
        promptAddedConnection = nil
    end
    if enableInstant then
        promptAddedConnection = workspace.DescendantAdded:Connect(function(descendant)
            if descendant:IsA("ProximityPrompt") then setupPrompt(descendant, true) end
        end)
    end
end

local function FindMarketToolId(ToolName)
    if not ReplicatedStorage:FindFirstChild("MarketItems") then return nil end
    for _, Item in ipairs(ReplicatedStorage.MarketItems:GetChildren()) do
        if Item.Name == ToolName and Item:FindFirstChild("owner") and Item.owner.Value == LocalPlayer.Name then
            return Item:GetAttribute("SpecialId") or Item:GetAttribute("ToolId")
        end
    end
end

local function GetUniqueToolNames()
    local ToolNames = {}
    if LocalPlayer.Character then
        local CurrentTool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
        if CurrentTool then ToolNames[CurrentTool.Name] = true end
    end
    local Results = {}
    for Name in pairs(ToolNames) do table.insert(Results, Name) end
    return Results
end

local function MarketProtectTool(ToolName)
    if MarketProtectedTools[ToolName] then return end
    local ToolId, Connection
    if ReplicatedStorage:FindFirstChild("MarketItems") then
        Connection = ReplicatedStorage.MarketItems.ChildAdded:Connect(function(Item)
            if Item.Name == ToolName and Item:FindFirstChild("owner") and Item.owner.Value == LocalPlayer.Name then
                ToolId = Item:GetAttribute("SpecialId")
            end
        end)
    end
    pcall(function()
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            LocalPlayer.Character:FindFirstChild("Humanoid"):UnequipTools()
        end
    end)
    task.wait(0.15)
    pcall(function()
        if ReplicatedStorage:FindFirstChild("ListWeaponRemote") then
            ReplicatedStorage.ListWeaponRemote:FireServer(ToolName, 676767)
        end
    end)
    task.wait(0.3)
    pcall(function()
        if ReplicatedStorage:FindFirstChild("BackpackRemote") then
            ReplicatedStorage.BackpackRemote:InvokeServer("Store", ToolName)
        end
    end)
    task.wait(3)
    if not ToolId then ToolId = FindMarketToolId(ToolName) end
    if Connection then Connection:Disconnect() end
    if ToolId then MarketProtectedTools[ToolName] = ToolId end
end

local function MarketReleaseTool(ToolName, ToolId)
    if not ToolName or not ToolId then return end
    if not ReplicatedStorage:FindFirstChild("BuyItemRemote") or not ReplicatedStorage:FindFirstChild("BackpackRemote") then return end
    pcall(function() ReplicatedStorage.BuyItemRemote:FireServer(ToolName, "Remove", ToolId) end)
    task.wait(0.5)
    pcall(function() ReplicatedStorage.BackpackRemote:InvokeServer("Grab", ToolName) end)
    MarketProtectedTools[ToolName] = nil
end

local function ProtectInventoryTools()
    for _, ToolName in ipairs(GetUniqueToolNames()) do MarketProtectTool(ToolName) end
end

local function ReleaseMarketTools()
    for ToolName, ToolId in pairs(MarketProtectedTools) do MarketReleaseTool(ToolName, ToolId) end
end

    end
end

local function getSelectedPlayer()
    local selected = PlayerUtilities.SelectedPlayer or flags["main_select_player"]
    selected = normalizeDropdownValue(selected)
    if typeof(selected) == "Instance" and selected:IsA("Player") then return selected end
    if type(selected) == "string" and selected ~= "" then
        local byName = Players:FindFirstChild(selected)
        if byName then return byName end

    return nil
end

local function getGunMuzzlePosition()
    local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("Handle") then
        local handle = tool.Handle
        local muzzle = handle:FindFirstChild("GunMuzzlePoint") or handle:FindFirstChild("GunFirePoint")
        if muzzle then return muzzle.WorldCFrame.Position end
        return handle.Position
    end
    return nil
end

local function getEquippedGunSetting()
    local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("Setting") then
        local ok, settings = pcall(require, tool.Setting)
        if ok and type(settings) == "table" then return tool, settings end
    end
    return nil, nil
end

local function applyAllWeaponMods()
    local tool, settings = getEquippedGunSetting()
    if not settings then return end

    pcall(function()
        if WeaponMods.InfiniteAmmo then
            settings.Ammo = 99999
            settings.AmmoPerMag = 99999
            settings.LimitedAmmoEnabled = false
        end
        if WeaponMods.InfiniteClips then
            settings.Ammo = 80000
            settings.AmmoPerMag = 80000
            settings.LimitedAmmoEnabled = false
        end
        if WeaponMods.Bullets80k then
            settings.Ammo = 80000
            settings.AmmoPerMag = 80000
            settings.LimitedAmmoEnabled = false
            local gunLocal = tool:FindFirstChild("GunScript_Local")
            if gunLocal and type(getsenv) == "function" then
                pcall(function()
                    local env = getsenv(gunLocal)
                    if env and type(env.Reload) == "function" and type(debug.setupvalue) == "function" then
                        debug.setupvalue(env.Reload, 1, 80000)
                        debug.setupvalue(env.Reload, 3, 80000)
                    end
                end)
            end
        end
        if WeaponMods.ModifyRecoilValue then
            settings.Recoil = 0
            settings.CameraRecoilingEnabled = false
        end
        if WeaponMods.ModifySpreadValue then
            settings.Spread = 0
            settings.SpreadX = 0
            settings.SpreadY = 0
            settings.Accuracy = 1
        end
        if WeaponMods.DisableJamming then
            settings.JamChance = 0
        end
        if WeaponMods.InstantReload then
            settings.ReloadTime = WeaponMods.ReloadSpeed or 0.1
        end
        if WeaponMods.InstantEquip then
            settings.EquipTime = WeaponMods.EquipSpeed or 0.1
            settings.EquippingTime = WeaponMods.EquipSpeed or 0.1
        end
        if WeaponMods.ModifyFireRate then
            settings.FireRate = 0
        end
        if WeaponMods.Automatic then
            settings.Auto = true
        end
        if WeaponMods.DamageAmplified then
            settings.BaseDamage = 10000
        end
        if WeaponMods.SpoofAccuracy then
            settings.Accuracy = 1
        end
    end)
end

local function wallCheckCharacter(character)
    if not character or not character:FindFirstChild("Head") then return false end
    local origin = Camera.CFrame.Position
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = { LocalPlayer.Character, Camera, character }
    params.FilterType = Enum.RaycastFilterType.Blacklist
    params.IgnoreWater = true
    return workspace:Raycast(origin, character.Head.Position - origin, params) == nil
end

local function isValidCombatPlayer(player)
    if not player or player == LocalPlayer then return false end
    local character = player.Character
    if not character then return false end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    return humanoid and humanoid.Health > 0 and root
end

local function getNearestPlayer(maxDistance)
    local localRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not localRoot then return nil end

    local nearest, nearestDistance = nil, maxDistance or math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if isValidCombatPlayer(player) then
            if RageBot.ProtectionCheck and player.Character:FindFirstChildOfClass("ForceField") then continue end
            local distance = (player.Character.HumanoidRootPart.Position - localRoot.Position).Magnitude
            if distance <= nearestDistance then
                nearestDistance = distance
                nearest = player
            end
        end
    end
    return nearest, nearestDistance
end

local function applyHitboxExpander()
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer or not player.Character then continue end
        local partName = RageBot.TargetBone == "Head" and "Head" or "HumanoidRootPart"
        local part = player.Character:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            if WeaponMods.HitboxExpander then
                local size = WeaponMods.HitboxSize * 2
                part.Size = Vector3.new(size, size, size)
                part.Transparency = WeaponMods.HitboxVisible and 0.5 or 1
                part.CanCollide = false
                part.Massless = true
            else
                part.Size = partName == "Head" and Vector3.new(2, 1, 1) or Vector3.new(2, 2, 1)
                part.Transparency = 1
            end
        end
    end
end

        end
    end
end)

RunService:BindToRenderStep("PlayerDesync", Enum.RenderPriority.Character.Value, function(dt)
    if PlayerDesync.SpectateNearest then
        desyncSpectateReset = false
        local target = getNearestPlayer()
        local subject = target and target.Character and target.Character:FindFirstChild("Humanoid")
        if subject then Camera.CameraSubject = subject end
    elseif not desyncSpectateReset and LocalPlayer.Character then
        desyncSpectateReset = true
        local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then Camera.CameraSubject = hum end
    end

    if not PlayerDesync.Enabled and not PlayerDesync.AttachNearest and not PlayerDesync.TweenNearest then
        return
    end

    local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local target = getNearestPlayer()
    if not target or not target.Character or not target.Character:FindFirstChild("HumanoidRootPart") then return end

    local targetRoot = target.Character.HumanoidRootPart

    if PlayerDesync.AttachNearest then
        root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
        return
    end

    if PlayerDesync.TweenNearest then
        root.CFrame = root.CFrame:Lerp(targetRoot.CFrame * CFrame.new(0, 0, 3), math.clamp(dt * 2, 0, 1))
        return
    end

    if PlayerDesync.Enabled then
        orbitAngle += dt * math.max(PlayerDesync.FlySpeed, 1)
        local offset
        if PlayerDesync.DesyncType == "Orbit" then
            offset = Vector3.new(math.cos(orbitAngle), 0.25, math.sin(orbitAngle)) * PlayerDesync.OrbitRadius
        else
            offset = Vector3.new(math.cos(orbitAngle), PlayerDesync.OrbitRadius * 0.5, math.sin(orbitAngle)) * math.max(PlayerDesync.OrbitRadius, 1)
        end
        root.CFrame = CFrame.new(targetRoot.Position + offset, targetRoot.Position)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end
end)

local function GetMousePosition()
    if Mouse and Mouse.X and Mouse.Y then return Vector2.new(Mouse.X, Mouse.Y) end
    return UserInputService:GetMouseLocation()
end

local function GetPartCollection(player, points)
    local collection = {}
    if not player or not player.Character then return collection end
    local pointMap = {
        head = {"Head"}, body = {"UpperTorso", "LowerTorso", "Torso", "HumanoidRootPart"}, arms = {"LeftUpperArm", "RightUpperArm", "LeftLowerArm", "RightLowerArm", "LeftHand", "RightHand"}, legs = {"LeftUpperLeg", "RightUpperLeg", "LeftLowerLeg", "RightLowerLeg", "LeftFoot", "RightFoot"}
    }
    local added = {}
    if type(points) == "string" then points = { points } end
    for _, point in ipairs(points or {"Head"}) do
        local names = pointMap[string.lower(point)] or {}
        for _, name in ipairs(names) do
            local part = player.Character:FindFirstChild(name)
            if part and not added[part] then added[part] = true; table.insert(collection, part) end
        end
    end
    if #collection == 0 then
        local fallback = player.Character:FindFirstChild("Head")
        if fallback then table.insert(collection, fallback) end
    end
    return collection
end

local function SelectTargetPart(player, points, priority)
    local parts = GetPartCollection(player, points)
    if #parts == 0 then return nil end
    if priority == "head" then
        for _, part in ipairs(parts) do if part.Name == "Head" then return part end end
    elseif priority == "body" then
        for _, part in ipairs(parts) do
            local n = part.Name
            if n == "HumanoidRootPart" or n == "Torso" or n == "UpperTorso" or n == "LowerTorso" then return part end
        end
    end
    return parts[math.random(1, #parts)]
end

local function IsPlayerValid(player)
    if not player or player == LocalPlayer then return false end
    local character = player.Character
    if not character then return false end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")
    return humanoid and humanoid.Health > 0 and root
end

local function ScreenDistanceToPart(part)
    if not part then return math.huge end
    local screenPoint, onScreen = Camera:WorldToViewportPoint(part.Position)
    if not onScreen then return math.huge end
    return (Vector2.new(screenPoint.X, screenPoint.Y) - GetMousePosition()).Magnitude
end

local function WallCheck(character)
    if not character then return false end
    local head = character:FindFirstChild("Head")
    if not head then return false end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Blacklist
    params.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    params.IgnoreWater = true
    return not workspace:Raycast(Camera.CFrame.Position, head.Position - Camera.CFrame.Position, params)
end

local function GetClosestPlayerToMouseAimAssist()
    if not flags["legit_aimassist"] then return nil end
    local bestTarget, bestDistance = nil, math.huge
    local points = flags["legit_aimassistpoints"] or {"Head"}
    local priority = flags["legit_aimassistpriority"] or "closest"
    local fov = 250
    for _, player in ipairs(Players:GetPlayers()) do
        if not IsPlayerValid(player) then continue end
        local targetPart = SelectTargetPart(player, points, priority)
        if not targetPart then continue end
        local distance = ScreenDistanceToPart(targetPart)
        if distance <= fov and distance < bestDistance then bestDistance = distance; bestTarget = player end
    end
    return bestTarget
end

local function GetClosestPlayerToMouseSilent()
    if not Silent.Enabled then return nil end
    local bestTarget, bestDistance = nil, math.huge
    local mousePos = GetMousePosition()
    local radius = Silent.UseFieldOfView and Silent.Radius or math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if not IsPlayerValid(player) then continue end
        local head = player.Character:FindFirstChild("Head")
        if not head then continue end
        local screenPos, onScreen = Camera:WorldToViewportPoint(head.Position)
        if not onScreen then continue end
        local distance = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
        if distance > radius then continue end
        if distance < bestDistance then bestDistance = distance; bestTarget = player end
    end
    return bestTarget
end

local function IsAimlockActive()
    if Aimlock.Activation == "always" then return true
    elseif Aimlock.Activation == "mouse 1" then return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    elseif Aimlock.Activation == "mouse 2" then return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)
    elseif Aimlock.Activation == "toggle" then return Aimlock.Aiming end
    return false
end

local function GetClosestPlayerToMouseAimbot()
    if not Aimlock.Enabled or not IsAimlockActive() then return nil end
    local bestTarget, bestDistance = nil, math.huge
    local mousePos = GetMousePosition()
    local radius = Aimlock.UseFieldOfView and Aimlock.Radius or 9e9
    for _, player in ipairs(Players:GetPlayers()) do
        if not IsPlayerValid(player) then continue end
        local targetPart = player.Character:FindFirstChild(Aimlock.TargetPart)
        if not targetPart then continue end
        if not playerDistanceCheck(player, Aimlock.MaxDistance) then continue end
        if Aimlock.WallCheck and not WallCheck(player.Character) then continue end
        local screenPoint, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
        if not onScreen then continue end
        local magnitude = (Vector2.new(screenPoint.X, screenPoint.Y) - mousePos).Magnitude
        if magnitude <= radius and magnitude < bestDistance then bestDistance = magnitude; bestTarget = player end
    end
    return bestTarget
end

local function IsAimAssistActive()
    local activation = flags["legit_aimassistactivation"]
    if activation == "always" then return true
    elseif activation == "mouse 1" then return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    elseif activation == "mouse 2" then return UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) end
    return false
end

local function isGunRaycastCall()
    if type(getcallingscript) ~= "function" then return true end
    local callingScript = getcallingscript()
    if not callingScript then return true end
    local n = callingScript.Name
    return n == "GunScript_Local" or n == "GunScript" or n == "BulletVisualizerServerScript" or n == "BulletVisualizerClientScript"
end

if hookmetamethod then
    local __namecall
    __namecall = hookmetamethod(workspace, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        local lowerMethod = string.lower(method)

        if checkcaller() then return __namecall(self, ...) end

        local isRaycast = string.find(lowerMethod, "raycast") ~= nil
        local isFindPartOnRay = string.find(lowerMethod, "findpartonray") ~= nil
        if not isRaycast and not isFindPartOnRay then return __namecall(self, ...) end
        if not isGunRaycastCall() then return __namecall(self, ...) end

        local args = {...}

        if Silent.Enabled then
            if not (math.random(0, 100) <= Silent.HitChance) then return __namecall(self, unpack(args)) end
            local target = SilentTarget
            if target and target.Character and IsPlayerValid(target) then
                local targetPart = target.Character:FindFirstChild(Silent.HitParts[1] or "Head") or target.Character:FindFirstChild("HumanoidRootPart") or target.Character:FindFirstChild("Head")
                if targetPart then
                    local origin
                    if isFindPartOnRay and args[1] then origin = args[1].Origin
                    elseif isRaycast and args[1] then origin = args[1] end

                    if origin then
                        local direction = (targetPart.Position - origin).Unit * 9e17
                        if isFindPartOnRay then
                            if Silent.WallBang then return targetPart, targetPart.Position, Vector3.new(0, 0, 0) end
                            args[1] = Ray.new(origin, direction)
                        elseif isRaycast then
                            args[2] = direction
                        end
                    end
                end
            end
        end
        return __namecall(self, unpack(args))
    end)
end

                local mousePos = GetMousePosition()
                local delta = Vector2.new(screenPoint.X - mousePos.X, screenPoint.Y - mousePos.Y)

                local smoothing = math.max(1, (flags["legit_aimassistsmoothing"] or 30) / 10)
                Move_Mouse_Function(delta.X / smoothing, delta.Y / smoothing)
            end
        end
    end

-- Fly System
local FlyCamera = workspace.CurrentCamera
local FlyRunService = game:GetService("RunService")
local flyEnabled = false
local flySwimMethod = false
local flySpeedConnection = nil
local flyCFloop = nil
local flySpeedVal = 5

local function flyHandleSpeed()
    if not flySwimMethod then return end
    local character = LocalPlayer.Character
    if not character or not character.Parent then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or not hrp or humanoid.Health <= 0 then return end
    humanoid:ChangeState(0)
    if flyEnabled then hrp.RotVelocity = Vector3.new(0, 0, 0) end
end

local function flyOnCharacterAdded(char)
    if flySpeedConnection then flySpeedConnection:Disconnect(); flySpeedConnection = nil end
    pcall(function() char:WaitForChild("Humanoid", 5); char:WaitForChild("HumanoidRootPart", 5) end)
    flySpeedConnection = FlyRunService.Stepped:Connect(function()
        local ok = pcall(flyHandleSpeed)
        if not ok and flySpeedConnection then flySpeedConnection:Disconnect(); flySpeedConnection = nil end
    end)
end

local function StartFly()
    if flyEnabled then return end
    flyEnabled = true; flySwimMethod = true; getgenv().SwimMethod = true
    task.wait(0.2)
    local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    if not humanoid or not head then return end
    humanoid.PlatformStand = true; head.Anchored = true
    if flyCFloop then flyCFloop:Disconnect() end
    flyCFloop = FlyRunService.Heartbeat:Connect(function(deltaTime)
        if not flyEnabled or not character or not character.Parent then if flyCFloop then flyCFloop:Disconnect() end; return end
        local moveDir = humanoid.MoveDirection
        local speed = flySpeedVal * deltaTime * 100
        local camCF = FlyCamera.CFrame
        local _, yRot, _ = camCF:ToOrientation()
        local uprightRot = CFrame.Angles(0, yRot, 0)
        local flatCamCF = CFrame.new(camCF.Position) * uprightRot
        local localInput = flatCamCF:VectorToObjectSpace(moveDir * speed)
        local worldMovement = camCF:VectorToWorldSpace(localInput)
        head.CFrame = CFrame.new(head.Position + worldMovement) * uprightRot
        FlyCamera.CameraSubject = head
    end)
end

local function StopFly()
    if not flyEnabled then return end
    flyEnabled = false; flySwimMethod = false; getgenv().SwimMethod = false
    if flyCFloop then flyCFloop:Disconnect(); flyCFloop = nil end
    local character = LocalPlayer.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local head = character:FindFirstChild("Head")
        if humanoid then humanoid.PlatformStand = false; humanoid:ChangeState(Enum.HumanoidStateType.Running) end
        if head then head.Anchored = false end
    end
end

    if choice == "None" then
        pcall(function()
            if originalSkySettings.ClockTime then Lighting_Sky.ClockTime = originalSkySettings.ClockTime end
            if originalSkySettings.Brightness then Lighting_Sky.Brightness = originalSkySettings.Brightness end
            if originalSkySettings.FogEnd then Lighting_Sky.FogEnd = originalSkySettings.FogEnd end

            if originalSkySettings.GlobalShadows then Lighting_Sky.GlobalShadows = originalSkySettings.GlobalShadows end
            if originalSkySettings.OutdoorAmbient then Lighting_Sky.OutdoorAmbient = originalSkySettings.OutdoorAmbient end

        atm.Color = preset.AtmColor; atm.Decay = preset.AtmDecay; atm.Glare = preset.AtmGlare; atm.Haze = preset.AtmHaze
    end)
end

local uiVisible = true
local function toggleUI()
    uiVisible = not uiVisible
    Window:SetOpen(uiVisible)
    textBadge.Text = uiVisible and "CLOSE" or "MENU"
    textBadge.BackgroundColor3 = uiVisible and Color3.fromRGB(18, 18, 22) or Color3.fromRGB(150, 20, 20)
end

toggleButton.MouseButton1Click:Connect(toggleUI)

local dragging, dragStart, startPos
toggleButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true; dragStart = input.Position; startPos = toggleButton.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        toggleButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Rainbow Discord Tag
local discordTagEnabled = true
local function removeDiscordTag(char)
    pcall(function()
        local head = char and char:FindFirstChild("Head")
        if head then
            local tag = head:FindFirstChild("OregonDiscordTag")
            if tag then tag:Destroy() end
        end
    end)
end

local function addDiscordTag(char)
    if not discordTagEnabled or not char then return end
    removeDiscordTag(char)

    local head = char:FindFirstChild("Head")
    if not head then task.wait(1); head = char:FindFirstChild("Head"); if not head then return end end

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "OregonDiscordTag"; billboard.Adornee = head; billboard.Size = UDim2.new(0, 200, 0, 30); billboard.StudsOffset = Vector3.new(0, 4, 0); billboard.AlwaysOnTop = true; billboard.Parent = head

    local text = Instance.new("TextLabel")
    text.Size = UDim2.new(1, 0, 1, 0); text.BackgroundTransparency = 1; text.Text = "discord.gg/oregonscripts"; text.TextStrokeTransparency = 0; text.TextScaled = true; text.Font = Enum.Font.GothamBold; text.Parent = billboard

    task.spawn(function()
        local hue = 0
        while billboard and billboard.Parent and discordTagEnabled do
            hue = (hue + 0.01) % 1; text.TextColor3 = Color3.fromHSV(hue, 1, 1); task.wait(0.03)
        end
    end)
end

if LocalPlayer.Character then task.spawn(function() addDiscordTag(LocalPlayer.Character) end) end
LocalPlayer.CharacterAdded:Connect(function(char) task.wait(1); addDiscordTag(char) end)
LocalPlayer.CharacterRemoving:Connect(removeDiscordTag)
